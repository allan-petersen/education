       ''            AS                                                                                        adtunittype,
       r.regioncode,
       CASE -- Hvis der allerede er en SHAK hospitals kode, saa anvendes denne, ellers anvendes den aftalte konstant
           WHEN r.hospitalcode IS NOT NULL THEN r.hospitalcode
           WHEN r.hospitalcode IS NULL AND r.id27_shortname IS NOT NULL THEN 'X999'
           ELSE NULL
           END       AS                                                                                        hospitalcode,
       CASE -- Hvis der allerede er en SHAK kode, anvendes denne ellers anvendes en af de aftalte konstanter
           WHEN r.shakidentifier IS NOT NULL THEN r.shakidentifier
           WHEN r.shakidentifier IS NULL AND r.id27_shortname = 22232009 THEN 'X999' --Hospital
           WHEN r.shakidentifier IS NULL AND r.id27_shortname = 568281000005108 THEN 'X99999' --Overafdeling
           WHEN r.shakidentifier IS NULL AND r.id27_shortname = 284548004 THEN 'X999999' --Afdeling
           ELSE NULL
           END       AS                                                                                        shakidentifier,
       e.locationcode                                                                                          eanlocationnumber,
       e_inherit.locationcode                                                                                  inheritedeanlocationnumber,
       r.municipalitycode,
       r.startdate,
       r.enddate,
       r.parentsoridentifier,
       r.parentmasterorgid,
       r.updated     AS                                                                                        updated,
       r.specialitycodes,
       r.shakspeciality,
       g.geographicallocalisationname,
       c.description AS                                                                                        sorEntityTypeName,
       c.shortname   AS                                                                                        sorEntityTypeIdentifier,
       r.providerIdentifier
FROM results r
         LEFT JOIN po_postaladdress p ON p.id = r.postaladdress_id
         LEFT JOIN sor_virtualaddress v ON v.id = r.virtualaddress_id
         LEFT JOIN sor_eanlocation e ON e.id = r.eanlocationnumber_id
         LEFT JOIN sor_eanlocation e_inherit ON e_inherit.id = r.inheritedeanlocationnumber_id
         LEFT JOIN sor_geographicallocalisation g ON g.geographicallocalisationidentifier = r.geographicallocalisationid
         LEFT JOIN cl_classid c ON c.id = r.sortype_classificationid AND c.systemid = r.sortype_systemid
--enheder uden sortype_classificationid sorteres fra, da de er gamle lukkede enheder der kun støjer
WHERE r.sortype_classificationid IS NOT NULL
ORDER BY NLSSORT(path, 'NLS_SORT=BINARY'), soridentifier
